cmake_minimum_required(VERSION 3.24)
project(DropscotchDiagnostics VERSION 1.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with the Windows x64 toolchain.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# The host exchanges std::string in PluginStatus: use the host's release DLL CRT.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
add_library(DropscotchDiagnostics SHARED Diagnostics.cpp)
target_include_directories(DropscotchDiagnostics PRIVATE sdk)
target_compile_definitions(DropscotchDiagnostics PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(DropscotchDiagnostics PRIVATE /W4 /EHsc /utf-8)
endif()
add_executable(diagnostic-smoke SmokeTest.cpp)
target_include_directories(diagnostic-smoke PRIVATE sdk)
target_compile_definitions(diagnostic-smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(diagnostic-smoke PRIVATE /W4 /EHsc /utf-8)
endif()
enable_testing()
add_test(NAME diagnostic-smoke COMMAND diagnostic-smoke $<TARGET_FILE:DropscotchDiagnostics>)
